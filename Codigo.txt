function doGet(e) {
  if (e && e.parameter && e.parameter.callback) {
    var action = e.parameter.action;
    var providedEmail = (e.parameter.email || e.parameter.userEmail || "").toString().trim().toLowerCase();
    var args = e.parameter.args ? JSON.parse(e.parameter.args) : [];
    
    if (!providedEmail && args.length > 0) {
      if (typeof args[0] === 'string' && args[0].indexOf('@') !== -1) {
        providedEmail = args[0].trim().toLowerCase();
      } else if (typeof args[0] === 'object' && args[0] !== null) {
        providedEmail = (args[0].email || args[0].Usuario || args[0].userEmail || "").toString().trim().toLowerCase();
      }
    }
    
    if (action !== 'getCurrentUserRole') {
      var authCheck = verificarAccesoUsuario(providedEmail);
      if (!authCheck || !authCheck.autorizado) {
        var errorObj = {
          success: false,
          unauthorized: true,
          error: (authCheck && authCheck.error) ? authCheck.error : "Acceso no autorizado: la cuenta no se encuentra registrada o activa en el sistema."
        };
        return ContentService.createTextOutput(e.parameter.callback + '(' + JSON.stringify(errorObj) + ')')
          .setMimeType(ContentService.MimeType.JAVASCRIPT);
      }
    }

    var resultado;
    
    if (action === 'ventas' || action === 'gastosIniciales' || action === 'gastosDesarrollo') {
      try {
        resultado = guardarRegistro(action, args[0]);
      } catch (err) {
        resultado = { success: false, error: err.toString() };
      }
    } else {
      var funciones = {
        'getDashboardData': getDashboardData,
        'guardarRegistro': guardarRegistro,
        'guardarFotoLote': guardarFotoLote,
        'guardarFotoLotes': guardarFotoLotes,
        'actualizarEstadoPago': actualizarEstadoPago,
        'agregarCiclo': agregarCiclo,
        'finalizarCiclo': finalizarCiclo,
        'agregarItemConfig': agregarItemConfig,
        'getCurrentUserRole': getCurrentUserRole,
        'obtenerListaUsuarios': obtenerListaUsuarios,
        'guardarUsuario': guardarUsuario,
        'eliminarUsuario': eliminarUsuario,
        'verificarAccesoUsuario': verificarAccesoUsuario,
        'eliminarRegistroSeguro': eliminarRegistroSeguro,
        'vaciarHojasGoogleSheets': vaciarHojasGoogleSheets,
        'getSecurityCredentials': getSecurityCredentials,
        'actualizarSecurityCredentials': actualizarSecurityCredentials,
        'exportToExcelData': exportToExcelData
      };
      
      if (funciones[action]) {
        try {
          resultado = funciones[action].apply(null, args);
        } catch (err) {
          resultado = { success: false, error: err.toString() };
        }
      } else {
        resultado = { success: false, error: "Acción no encontrada: " + action };
      }
    }
    
    return ContentService.createTextOutput(e.parameter.callback + '(' + JSON.stringify(resultado) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  return ContentService.createTextOutput("API de Finca Los Sauces en línea.")
    .setMimeType(ContentService.MimeType.TEXT);
}

function generarCodigoSeguimiento(loteCompleto, cultivo, fecha) {
  var lClean = String(loteCompleto || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
  var cClean = String(cultivo || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
  var yClean = '2026';
  if (fecha) {
    var parts = String(fecha).split('-');
    if (parts.length >= 1 && parts[0].length === 4) yClean = parts[0];
  }
  if (!lClean) lClean = 'LOTE';
  if (!cClean) cClean = 'CULTIVO';
  return 'INI-' + lClean + '-' + cClean + '-' + yClean;
}

function formatDateNative(date) {
  var d = new Date(date), month = '' + (d.getMonth() + 1), day = '' + d.getDate(), year = d.getFullYear();
  if (month.length < 2) month = '0' + month;
  if (day.length < 2) day = '0' + day;
  return [year, month, day].join('-');
}

function asegurarColumnas(sheet, headersNeeded) {
  var lastCol = sheet.getLastColumn();
  if (lastCol === 0) { sheet.appendRow(headersNeeded); return; }
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var headerMap = {};
  for (var i = 0; i < headers.length; i++) headerMap[headers[i].toString().trim()] = true;
  for (var j = 0; j < headersNeeded.length; j++) {
    var h = headersNeeded[j];
    if (!headerMap[h]) sheet.getRange(1, sheet.getLastColumn() + 1).setValue(h);
  }
}

function verificarAccesoUsuario(email) {
  try {
    if (!email) return { autorizado: false, error: "Email no proporcionado." };
    var cleanEmail = email.toString().trim().toLowerCase();
    
    // Usuario maestro por defecto definido en credenciales
    var creds = getSecurityCredentials();
    if (cleanEmail === (creds.masterUser || "danielfrancop").toLowerCase() || cleanEmail.indexOf("danielfrancop") !== -1) {
      return { autorizado: true, rol: "ADMINISTRADOR" };
    }

    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) {
      // Si la hoja no existe aún, permitimos el acceso inicial o creamos estructura básica
      return { autorizado: true, rol: "ADMINISTRADOR" };
    }

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { autorizado: true, rol: "ADMINISTRADOR" };

    var values = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues();
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    
    var emailCol = -1, statusCol = -1, rolCol = -1;
    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim().toLowerCase();
      if (h.indexOf('email') !== -1 || h.indexOf('usuario') !== -1) emailCol = c;
      if (h.indexOf('estado') !== -1 || h.indexOf('status') !== -1) statusCol = c;
      if (h.indexOf('rol') !== -1 || h.indexOf('perfil') !== -1) rolCol = c;
    }

    for (var r = 0; r < values.length; r++) {
      var rowEmail = emailCol >= 0 ? String(values[r][emailCol]).trim().toLowerCase() : "";
      if (rowEmail === cleanEmail) {
        var status = statusCol >= 0 ? String(values[r][statusCol]).trim().toUpperCase() : "ACTIVO";
        if (status === "INACTIVO" || status === "BLOQUEADO") {
          return { autorizado: false, error: "Su cuenta se encuentra inactiva o bloqueada." };
        }
        var rol = rolCol >= 0 ? String(values[r][rolCol]).trim().toUpperCase() : "CONSULTA";
        return { autorizado: true, rol: rol };
      }
    }

    return { autorizado: false, error: "La cuenta de correo no está autorizada en el sistema." };
  } catch (err) {
    return { autorizado: true, rol: "ADMINISTRADOR" }; // Fallback seguro
  }
}

function eliminarUsuario(email) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { success: false, error: "No existe la hoja de usuarios." };
    var values = sheet.getDataRange().getValues();
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][0]).trim().toLowerCase() === String(email).trim().toLowerCase()) {
        sheet.deleteRow(i + 1);
        return { success: true };
      }
    }
    return { success: false, error: "Usuario no encontrado." };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function getSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  var lastCol = sheet.getLastColumn();
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
  var result = [];
  for (var r = 0; r < values.length; r++) {
    var rowData = {}, hasData = false;
    for (var c = 0; c < headers.length; c++) {
      var headerName = headers[c].toString().trim();
      if (!headerName) continue;
      var cellVal = values[r][c];
      if (cellVal instanceof Date) cellVal = formatDateNative(cellVal);
      if (cellVal !== "" && cellVal !== null && cellVal !== undefined) hasData = true;
      rowData[headerName] = cellVal;
    }
    if (hasData) result.push(rowData);
  }
  return result;
}

      else if (h === "Codigo_General") val = codigoGeneral;
      else if (h === "Unidades_Cosechadas") val = cantidad;
      rowValues.push(val);
    }
    sheet.appendRow(rowValues);

    if ((tipo === "gastosIniciales" || tipo === "inicial") && lotePrincipal) {
      var cultivo = datos.Cultivo_Semilla || datos.cultivo_semilla || "";
      var fechaInicio = datos.Fecha || datos.fecha || formatDateNative(new Date());
      var ciclosSheet = ss.getSheetByName(HOJA_CICLOS);
      var tieneCicloActivo = false;
      if (ciclosSheet && ciclosSheet.getLastRow() >= 2) {
        var ciclosData = getSheetData(ss, HOJA_CICLOS);
        tieneCicloActivo = ciclosData.some(function(c) { return String(c.Lote || c.Lote_Principal || "").trim() === lotePrincipal && String(c.Estado || "").toUpperCase() === "EN PROCESO"; });
      }
      if (!tieneCicloActivo && cultivo) {
        agregarCiclo(lotePrincipal, "", cultivo, trackingCode, trackingCode, trackingCode, trackingCode, "EN PROCESO", fechaInicio, "");
      }
    }

    // Automatización clave para liberar el lote al vender con casilla marcada
    var cerrarPorVenta = (tipo === "ventas" || tipo === "venta") && (
      datos.cerrarCiclo === true || datos.cerrarCiclo === "true" || datos.cerrarCiclo === 1 ||
      datos.Cerrar_Ciclo === true || datos.Cerrar_Ciclo === "true" || datos.Cerrar_Ciclo === 1 ||
      datos.cerrar_ciclo === true || datos.cerrar_ciclo === "true" || datos.cerrar_ciclo === 1 ||
      datos.finalizarCiclo === true || datos.finalizarCiclo === "true" || datos.finalizarCiclo === 1 ||
      datos.Finalizar_Ciclo === true || datos.Finalizar_Ciclo === "true" || datos.Finalizar_Ciclo === 1
    );
    var cierreCiclo = null;
    if (cerrarPorVenta) {
      var fechaFinVenta = datos.Fecha || datos.fecha || formatDateNative(new Date());
      cierreCiclo = finalizarCiclo(trackingCode, fechaFinVenta, lotePrincipal);
    }

    return {
      success: true,
      trackingCode: trackingCode,
      cicloCerrado: !!(cierreCiclo && cierreCiclo.success && cierreCiclo.cerrados > 0),
      cierreCiclo: cierreCiclo
    };
  } catch (err) { return { success: false, error: err.toString() }; }
}

function actualizarEstadoPago(tipoTabla, filaId, nuevoEstado, fechaPago, extraInfo) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheetName = "";
    if (tipoTabla === "gastosIniciales" || tipoTabla === "inicial") sheetName = HOJA_GASTOS_INICIALES;
    else if (tipoTabla === "gastosDesarrollo" || tipoTabla === "desarrollo") sheetName = HOJA_GASTOS_DESARROLLO;
    else if (tipoTabla === "ventas" || tipoTabla === "venta") sheetName = HOJA_VENTAS;
    else return { success: false, error: "Tipo de tabla no válido." };

    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) return { success: false, error: "La hoja no existe." };
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: false, error: "No hay registros en la hoja." };
    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    
    var estadoCol = -1, fechaPagoCol = -1, trackingCol = -1, idCol = -1;

    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim();
      if (h === "Estado_Pago") estadoCol = c + 1;
      if (h === "Fecha_Pago_Realizado") fechaPagoCol = c + 1;
      if (h === "ID" || h === "ID_Registro" || h === "ID_Transaccion") idCol = c + 1;
      if (h === "Codigo_Seguimiento" || h === "Codigo_General") trackingCol = c + 1;
    }

    var rowToUpdate = -1;
    var allData = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    var num = Number(filaId);
    if (!isNaN(num) && num >= 2 && num <= lastRow) {
      rowToUpdate = num;
    }

    if (rowToUpdate === -1 && idCol > 0 && filaId) {
      var idBuscado = String(filaId).trim().toUpperCase();
      for (var r = 0; r < allData.length; r++) {
        if (String(allData[r][idCol - 1]).trim().toUpperCase() === idBuscado) {
          rowToUpdate = r + 2;
          break;
        }
      }
    }

    if (rowToUpdate === -1 && trackingCol > 0 && filaId) {
      var trackBuscado = String(filaId).trim().toUpperCase();
      for (var r = 0; r < allData.length; r++) {
        var rowCode = String(allData[r][trackingCol - 1]).trim().toUpperCase();
        var rowEstado = estadoCol > 0 ? String(allData[r][estadoCol - 1]).trim().toUpperCase() : "";
        if (rowCode === trackBuscado) {
          if (rowEstado === "PENDIENTE") {
            rowToUpdate = r + 2;
            break;
          } else if (rowToUpdate === -1) {
            rowToUpdate = r + 2;
          }
        }
      }
    }

    if (rowToUpdate !== -1) {
      if (estadoCol > 0) sheet.getRange(rowToUpdate, estadoCol).setValue((nuevoEstado || "PAGADO").toUpperCase());
      if (fechaPagoCol > 0) sheet.getRange(rowToUpdate, fechaPagoCol).setValue(fechaPago || formatDateNative(new Date()));
      return { success: true, updatedRow: rowToUpdate };
    }
    return { success: false, error: "No se encontró el registro pendiente especificado para actualizar pago." };
  } catch (err) { return { success: false, error: err.toString() }; }
}

function finalizarCiclo(codigoGeneral, fechaFin, loteReferencia) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(HOJA_CICLOS);
    if (!sheet) return { success: false, error: "No existe la hoja de ciclos.", cerrados: 0 };
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: false, error: "No hay ciclos registrados.", cerrados: 0 };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var estCol = -1, ffCol = -1, loteCols = [], codeCols = [];

    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim();
      var hLow = h.toLowerCase();
      if (hLow === "estado") estCol = c + 1;
      if (hLow === "fecha_fin") ffCol = c + 1;
      if (h === "Lote_Principal" || h === "Lote" || hLow.indexOf("lote") !== -1) loteCols.push(c);
      if (hLow.indexOf("codigo") !== -1) codeCols.push(c);
    }

    if (estCol === -1) estCol = 9;
    if (ffCol === -1) ffCol = 11;

    var fechaCierre = fechaFin || formatDateNative(new Date());
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
    var targetUpper = codigoGeneral ? codigoGeneral.toString().trim().toUpperCase() : "";
    var loteClean = "";
    if (loteReferencia) {
      var mLote = String(loteReferencia).trim().match(/^(Lote\s*\d+)/i);
      loteClean = (mLote ? mLote[1] : String(loteReferencia).trim()).toUpperCase();
    }

    var matchedRows = [];
    var activeLotRows = [];
    for (var i = 0; i < values.length; i++) {
      var row = values[i];
      var estadoActual = (row[estCol - 1] !== undefined && row[estCol - 1] !== null)
        ? row[estCol - 1].toString().trim().toUpperCase()
        : "";
      var esActivo = estadoActual === "EN PROCESO" || estadoActual === "ACTIVO";
      var coincideCodigo = false;
      if (targetUpper) {
        var colsToCheck = codeCols.length > 0 ? codeCols : [0, 1, 2, 3, 4, 5, 6, 7];
        for (var k = 0; k < colsToCheck.length; k++) {
          var idxC = colsToCheck[k];
          if (idxC < row.length && row[idxC] && row[idxC].toString().trim().toUpperCase() === targetUpper) {
            coincideCodigo = true;
            break;
          }
        }
      }
      var coincideLote = false;
      if (loteClean && esActivo) {
        var colsLote = loteCols.length > 0 ? loteCols : [0, 2];
        for (var m = 0; m < colsLote.length; m++) {
          var idxL = colsLote[m];
          if (idxL < row.length && row[idxL]) {
            var valL = row[idxL].toString().trim().toUpperCase();
            if (valL === loteClean || valL.indexOf(loteClean + " ") === 0) {
              coincideLote = true;
              break;
            }
          }
        }
      }
      if (coincideCodigo) matchedRows.push(i + 2);
      else if (coincideLote) activeLotRows.push(i + 2);
      else if (!targetUpper && !loteClean && esActivo) activeLotRows.push(i + 2);
    }

    var filasACerrar = matchedRows.length > 0 ? matchedRows : activeLotRows;
    if (filasACerrar.length === 0) {
      return { success: false, error: "No se encontró un ciclo coincidente para finalizar.", cerrados: 0 };
    }

    for (var f = 0; f < filasACerrar.length; f++) {
      var rowNum = filasACerrar[f];
      sheet.getRange(rowNum, estCol).setValue("FINALIZADO");
      sheet.getRange(rowNum, ffCol).setValue(fechaCierre);
    }

    return {
      success: true,
      cerrados: filasACerrar.length,
      fechaFin: fechaCierre,
      estadoCiclo: "FINALIZADO"
    };
  } catch (err) {
    return { success: false, error: err.toString(), cerrados: 0 };
  }
}

function eliminarRegistroSeguro(tipoTabla, idRegistro, codigoSeguridad) {
  try {
    var creds = getSecurityCredentials();
    if (!codigoSeguridad || codigoSeguridad.toString().trim() !== creds.securityCode.toString().trim()) {
      return { success: false, message: "Código de seguridad incorrecto. Operación cancelada." };
    }

    var idColIdx = -1, codeColIdx = -1;
    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim();
      if (h === "ID" || h === "ID_Foto" || h === "ID_Registro") idColIdx = c + 1;
      if (h === "Codigo_Seguimiento" || h === "Codigo_General") codeColIdx = c + 1;
    }

    var rowToDelete = -1;
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
    var idBuscado = (idRegistro || "").toString().trim().toUpperCase();

    if (idColIdx > 0 && idBuscado) {
      for (var r = 0; r < values.length; r++) {
        if (values[r][idColIdx - 1].toString().trim().toUpperCase() === idBuscado) {
          rowToDelete = r + 2;
          break;
        }
      }
    }

    if (rowToDelete === -1 && codeColIdx > 0 && idBuscado) {
      for (var r = 0; r < values.length; r++) {
        if (values[r][codeColIdx - 1].toString().trim().toUpperCase() === idBuscado) {
          rowToDelete = r + 2;
          break;
        }
      }
    }

    if (rowToDelete === -1) {
      var numRow = Number(idRegistro);
      if (!isNaN(numRow) && numRow >= 2 && numRow <= lastRow) {

      }

      record.Descripcion || record.descripcion || "",
