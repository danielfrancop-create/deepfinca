var HOJA_USUARIOS = "Usuarios_Autorizados";
var HOJA_USUARIOS_LEGACY = "Usuarios";

function _getAuthSpreadsheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss && typeof SPREADSHEET_ID !== "undefined") ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  if (!ss) throw new Error("No se pudo acceder a la hoja de usuarios.");
  return ss;
}

function _getAuthUserSheet_(ss) {
  var names = [HOJA_USUARIOS, "Usuarios_Autorizados", "usuarios", "Usuarios"];
  for (var i = 0; i < names.length; i++) {
    var sheet = ss.getSheetByName(names[i]);
    if (sheet && sheet.getLastRow() >= 2) return sheet;
  }
  throw new Error("No hay usuarios autorizados configurados. Agrega manualmente un administrador activo.");
}

function _getAuthenticatedUser_() {
  var email = "";
  try {
    if (typeof Session !== "undefined" && Session.getActiveUser) email = String(Session.getActiveUser().getEmail() || "").trim().toLowerCase();
  } catch (err) {
    throw new Error("No se pudo verificar la identidad de Google.");
  }
  if (!email) throw new Error("Google Apps Script no proporcionó una identidad autenticada. Configura la aplicación para ejecutarse como el usuario que accede.");

  var sheet = _getAuthUserSheet_(_getAuthSpreadsheet_());
  var lastColumn = sheet.getLastColumn();
  if (lastColumn < 1) throw new Error("La hoja de usuarios no tiene encabezados válidos.");
  var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0].map(function(value) { return String(value || "").trim().toLowerCase(); });
  var emailColumn = headers.findIndex(function(value) { return ["correo_electronico", "email", "correo", "usuario"].indexOf(value) !== -1; });
  var roleColumn = headers.findIndex(function(value) { return ["rol", "role"].indexOf(value) !== -1; });
  var statusColumn = headers.findIndex(function(value) { return ["estado", "status"].indexOf(value) !== -1; });
  if (emailColumn < 0 || roleColumn < 0 || statusColumn < 0) throw new Error("La hoja de usuarios debe tener columnas de correo, rol y estado.");

  var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, lastColumn).getValues();
  for (var r = 0; r < values.length; r++) {
    var row = values[r];
    if (String(row[emailColumn] || "").trim().toLowerCase() !== email) continue;
    var role = String(row[roleColumn] || "").trim().toLowerCase();
    var status = String(row[statusColumn] || "").trim().toUpperCase();
    if (status !== "ACTIVO" && status !== "ACTIVE") throw new Error("La cuenta está inactiva o bloqueada.");
    if (role !== "admin" && role !== "view") throw new Error("La cuenta no tiene un rol válido asignado.");
    var permission = function(primary, fallback) {
      var index = headers.indexOf(primary);
      if (index < 0 && fallback) index = headers.indexOf(fallback);
      if (index < 0 || row[index] === "" || row[index] === null) return role === "admin" ? "Sí" : "No";
      return String(row[index]).trim();
    };
    return {
      email: email,
      role: role,
      active: true,
      permissions: {
        Modulo_Dashboard: permission("modulo_dashboard", "modulo_inicio"),
        Modulo_POS: permission("modulo_pos", "modulo_gastos"),
        Modulo_MapaLotes: permission("modulo_mapalotes", "modulo_lotes"),
        Modulo_AnalisisSeguimiento: permission("modulo_analisisseguimiento", "modulo_dashboard")
      }
    };
  }
  throw new Error("La identidad autenticada no está autorizada.");
}

function _requireAuthenticatedUser_() {
  return _getAuthenticatedUser_();
}

function _requireAdminUser_() {
  var user = _requireAuthenticatedUser_();
  if (user.role !== "admin") throw new Error("Acceso denegado. Se requiere rol administrador.");
  return user;
}

function _hasUserModuleAccess_(user, moduleName) {
  if (user.role === "admin") return true;
  var value = String((user.permissions || {})[moduleName] || "").trim().toLowerCase();
  return ["sí", "si", "true", "1", "activo"].indexOf(value) !== -1;
}

function _requireAnyModuleUser_(moduleNames) {
  var user = _requireAuthenticatedUser_();
  if (user.role === "admin") return user;
  if (!Array.isArray(moduleNames) || !moduleNames.some(function(name) { return _hasUserModuleAccess_(user, name); })) {
    throw new Error("Acceso denegado al módulo solicitado.");
  }
  return user;
}

function _requireModuleUser_(moduleName) {
  return _requireAnyModuleUser_([moduleName]);
}

function getCurrentUserRole() {
  try {
    var user = _getAuthenticatedUser_();
    return { success: true, registered: true, active: true, email: user.email, role: user.role, permissions: user.permissions };
  } catch (err) {
    return { success: false, registered: false, active: false, email: "", role: "view", permissions: { Modulo_Dashboard: "No", Modulo_POS: "No", Modulo_MapaLotes: "No", Modulo_AnalisisSeguimiento: "No" }, error: err.message };
  }
}

function verificarAccesoUsuario() {
  try {
    var user = _requireAuthenticatedUser_();
    return { autorizado: true, email: user.email, role: user.role, permissions: user.permissions };
  } catch (err) {
    return { autorizado: false, email: "", role: "denied", permissions: {}, error: err.message };
  }
}

function _getUserColumns_(headers) {
  var lower = headers.map(function(value) { return String(value || "").trim().toLowerCase(); });
  var find = function(names) { return lower.findIndex(function(value) { return names.indexOf(value) !== -1; }); };
  return {
    email: find(["correo_electronico", "email", "correo", "usuario"]),
    name: find(["nombre", "name"]),
    role: find(["rol", "role"]),
    status: find(["estado", "status"]),
    id: find(["id", "id_usuario"]),
    dashboard: find(["modulo_dashboard", "modulo_inicio"]),
    pos: find(["modulo_pos", "modulo_gastos"]),
    lots: find(["modulo_mapalotes", "modulo_lotes"])
  };
}

function obtenerListaUsuarios() {
  try {
    _requireAdminUser_();
    var sheet = _getAuthUserSheet_(_getAuthSpreadsheet_());
    var lastColumn = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
    var columns = _getUserColumns_(headers);
    if (columns.email < 0 || columns.role < 0 || columns.status < 0) throw new Error("La hoja de usuarios no tiene las columnas requeridas.");
    var values = sheet.getLastRow() > 1 ? sheet.getRange(2, 1, sheet.getLastRow() - 1, lastColumn).getValues() : [];
    var data = values.map(function(row) {
      return {
        id: columns.id >= 0 ? row[columns.id] : "",
        email: row[columns.email],
        nombre: columns.name >= 0 ? row[columns.name] : "",
        rol: String(row[columns.role] || "view").trim().toLowerCase(),
        estado: String(row[columns.status] || "INACTIVO").trim().toUpperCase(),
        moduloDashboard: columns.dashboard >= 0 ? row[columns.dashboard] : "No",
        moduloPOS: columns.pos >= 0 ? row[columns.pos] : "No",
        moduloMapaLotes: columns.lots >= 0 ? row[columns.lots] : "No"
      };
    }).filter(function(user) { return String(user.email || "").trim() !== ""; });
    return { success: true, data: data };
  } catch (err) { return { success: false, error: err.message || String(err) }; }
}

function guardarUsuario(datos) {
  try {
    var actor = _requireAdminUser_();
    if (typeof datos === "string") datos = JSON.parse(datos);
    if (!datos || typeof datos !== "object") throw new Error("Los datos del usuario no son válidos.");
    var email = String(datos.email || datos.Email || datos.Correo_Electronico || "").trim().toLowerCase();
    var role = String(datos.rol || datos.Rol || "view").trim().toLowerCase();
    var status = String(datos.estado || datos.Estado || "ACTIVO").trim().toUpperCase();
    if (!email || (role !== "admin" && role !== "view") || (status !== "ACTIVO" && status !== "INACTIVO")) throw new Error("Correo, rol o estado inválido.");
    if (email === actor.email && (role !== "admin" || status !== "ACTIVO")) throw new Error("No puedes quitarte el rol de administrador ni desactivar tu propia cuenta.");

    var sheet = _getAuthUserSheet_(_getAuthSpreadsheet_());
    var lastColumn = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
    var columns = _getUserColumns_(headers);
    if (columns.email < 0 || columns.role < 0 || columns.status < 0) throw new Error("La hoja de usuarios no tiene las columnas requeridas.");
    var values = sheet.getLastRow() > 1 ? sheet.getRange(2, 1, sheet.getLastRow() - 1, lastColumn).getValues() : [];
    var targetRow = -1;
    for (var i = 0; i < values.length; i++) {
      if (String(values[i][columns.email] || "").trim().toLowerCase() === email) { targetRow = i + 2; break; }
    }
    var row = targetRow > 0 ? values[targetRow - 2].slice() : new Array(lastColumn).fill("");
    var set = function(index, value) { if (index >= 0) row[index] = value; };
    set(columns.email, email);
    set(columns.name, datos.nombre || datos.Nombre || email.split("@")[0]);
    set(columns.role, role);
    set(columns.status, status);
    set(columns.dashboard, datos.Modulo_Dashboard || datos.moduloDashboard || "No");
    set(columns.pos, datos.Modulo_POS || datos.moduloPOS || "No");
    set(columns.lots, datos.Modulo_MapaLotes || datos.moduloMapaLotes || "No");
    if (columns.id >= 0 && !row[columns.id]) row[columns.id] = "USR-" + new Date().getTime();
    if (targetRow > 0) sheet.getRange(targetRow, 1, 1, lastColumn).setValues([row]);
    else sheet.appendRow(row);
    return { success: true, message: "Usuario guardado con éxito." };
  } catch (err) { return { success: false, error: err.message || String(err) }; }
}

function eliminarUsuario(email) {
  try {
    var actor = _requireAdminUser_();
    var target = String(email || "").trim().toLowerCase();
    if (!target) throw new Error("Especifica un correo válido.");
    if (target === actor.email) throw new Error("No puedes eliminar tu propia cuenta de administrador.");
    var sheet = _getAuthUserSheet_(_getAuthSpreadsheet_());
    var lastColumn = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
    var columns = _getUserColumns_(headers);
    if (columns.email < 0 || columns.role < 0 || columns.status < 0) throw new Error("La hoja de usuarios no tiene las columnas requeridas.");
    var values = sheet.getLastRow() > 1 ? sheet.getRange(2, 1, sheet.getLastRow() - 1, lastColumn).getValues() : [];
    var targetRow = -1;
    var activeAdmins = 0;
    var targetIsAdmin = false;
    for (var i = 0; i < values.length; i++) {
      var rowEmail = String(values[i][columns.email] || "").trim().toLowerCase();
      var role = String(values[i][columns.role] || "").trim().toLowerCase();
      var status = String(values[i][columns.status] || "").trim().toUpperCase();
      if (rowEmail === target) { targetRow = i + 2; targetIsAdmin = role === "admin" && (status === "ACTIVO" || status === "ACTIVE"); }
      if (role === "admin" && (status === "ACTIVO" || status === "ACTIVE")) activeAdmins++;
    }
    if (targetRow < 0) throw new Error("No se encontró el usuario solicitado.");
    if (targetIsAdmin && activeAdmins <= 1) throw new Error("No se puede eliminar el último administrador activo.");
    sheet.deleteRow(targetRow);
    return { success: true, message: "Usuario eliminado correctamente." };
  } catch (err) { return { success: false, error: err.message || String(err) }; }
}

  "ID", 
  "Correo_Electronico", 
  "Nombre", 
  "Rol", 
  "Estado", 
  "Modulo_Dashboard", 
  "Modulo_POS", 
  "Modulo_MapaLotes"
];

/**
 * Función que verifica la existencia de la hoja Usuarios_Autorizados y determina el rol y permisos por módulo del usuario activo.
 * @param {string} [providedEmail] Correo opcional enviado desde la vista para verificación
 * @return {Object} Objeto JSON con success, registered, active, email, role y permissions por módulo
 */
function legacyGetCurrentUserRole_(providedEmail) {
  return getCurrentUserRole();
  var defaultNoPerms = {
    Modulo_Dashboard: "No",
    Modulo_POS: "No",
    Modulo_MapaLotes: "No"
  };

  var defaultAdminPerms = {
    Modulo_Dashboard: "Sí",
    Modulo_POS: "Sí",
    Modulo_MapaLotes: "Sí"
  };

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      return {
        success: false,
        registered: false,
        active: false,
        email: "",
        role: "view",
        permissions: defaultNoPerms,
        error: "No se pudo obtener la hoja de cálculo activa."
      };
    }

    var sheet = ss.getSheetByName(HOJA_USUARIOS);

    if (!sheet) {
      // Migración si existe la hoja Usuarios legacy
      var oldSheet = ss.getSheetByName(HOJA_USUARIOS_LEGACY);
      if (oldSheet) {
        try { oldSheet.setName(HOJA_USUARIOS); sheet = oldSheet; } catch(e) {}
      }
      if (!sheet) {
        sheet = ss.insertSheet(HOJA_USUARIOS);
        sheet.appendRow(HEADERS_USUARIOS);
      }
    }
    if (typeof asegurarColumnas === "function") {
      asegurarColumnas(sheet, HEADERS_USUARIOS);
    }

    // 1. Obtener correo del usuario activo de Google
    var activeEmail = "";
    try {
      if (typeof Session !== "undefined" && Session.getActiveUser) {
        activeEmail = Session.getActiveUser().getEmail();
      }
    } catch (e) { activeEmail = ""; }

    var effectiveEmail = "";
    try {
      if (typeof Session !== "undefined" && Session.getEffectiveUser) {
        effectiveEmail = Session.getEffectiveUser().getEmail();
      }
    } catch (e) { effectiveEmail = ""; }

    var userEmail = "";
    if (providedEmail && typeof providedEmail === "string" && providedEmail.trim()) {
      userEmail = providedEmail.trim().toLowerCase();
    } else if (activeEmail) {
      userEmail = activeEmail.trim().toLowerCase();
    } else if (effectiveEmail) {
      userEmail = effectiveEmail.trim().toLowerCase();
    }

    var lastRow = sheet.getLastRow();

    // Si la hoja Usuarios_Autorizados está vacía (solo encabezado), inicializar con el usuario como primer Administrador
    if (lastRow < 2) {
      var initialAdmin = userEmail || activeEmail || effectiveEmail || "admin@fincasauces.com";
      initialAdmin = initialAdmin.trim().toLowerCase();
      sheet.appendRow(["USR-001", initialAdmin, initialAdmin.split('@')[0], "admin", "ACTIVO", "Sí", "Sí", "Sí", "Sí", "Sí", "Sí"]);
      return {
        success: true,
        registered: true,
        active: true,
        email: initialAdmin,
        role: "admin",
        permissions: defaultAdminPerms
      };
    }

    // Si no se detectó correo
    if (!userEmail) {
      return {
        success: true,
        registered: false,
        active: false,
        email: "",
        role: "view",
        permissions: defaultNoPerms,
        error: "NO_EMAIL_DETECTED"
      };
    }

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    var colMap = {};
    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim().toLowerCase();
      colMap[h] = c;
    }

    var emailColIdx = colMap["correo_electronico"] !== undefined ? colMap["correo_electronico"] : (colMap["email"] !== undefined ? colMap["email"] : 1);
    var roleColIdx = colMap["rol"] !== undefined ? colMap["rol"] : (colMap["role"] !== undefined ? colMap["role"] : 3);
    var statusColIdx = colMap["estado"] !== undefined ? colMap["estado"] : (colMap["status"] !== undefined ? colMap["status"] : 4);

    var foundUser = null;
    for (var r = 0; r < values.length; r++) {
      var row = values[r];
      var rowEmail = row[emailColIdx] ? row[emailColIdx].toString().trim().toLowerCase() : "";
      if (rowEmail === userEmail) {
        var role = row[roleColIdx] ? row[roleColIdx].toString().trim().toLowerCase() : "view";
        var status = row[statusColIdx] ? row[statusColIdx].toString().trim().toUpperCase() : "INACTIVO";
        var isA = (role === "admin");

        var getPerm = function(primaryKey, fallbackKey) {
          if (colMap[primaryKey] !== undefined && row[colMap[primaryKey]] !== undefined && row[colMap[primaryKey]] !== "") {
            return row[colMap[primaryKey]].toString().trim();
          }
          if (fallbackKey && colMap[fallbackKey] !== undefined && row[colMap[fallbackKey]] !== undefined && row[colMap[fallbackKey]] !== "") {
            return row[colMap[fallbackKey]].toString().trim();
          }
          return isA ? "Sí" : "Sí";
        };

        var perms = {
          Modulo_Dashboard: getPerm("modulo_dashboard", "modulo_inicio"),
          Modulo_POS: getPerm("modulo_pos", "modulo_gastos"),
          Modulo_MapaLotes: getPerm("modulo_mapalotes", "modulo_lotes")
        };

        foundUser = {
          email: rowEmail,
          role: isA ? "admin" : "view",
          status: status,
          permissions: perms
        };
        break;
      }
    }

    if (!foundUser) {
      return {
        success: true,
        registered: false,
        active: false,
        email: userEmail,
        role: "view",
        permissions: defaultNoPerms,
        error: "Acceso denegado: El correo " + userEmail + " no está en la lista blanca de Usuarios_Autorizados."
      };
    }

    var isActive = (foundUser.status === "ACTIVO" || foundUser.status === "ACTIVE");

    return {
      success: true,
      registered: true,
      active: isActive,
      email: foundUser.email,
      role: foundUser.role,
      permissions: foundUser.permissions
    };

  } catch (err) {
    return {
      success: false,
      registered: false,
      active: false,
      email: "",
      role: "view",
      permissions: defaultNoPerms,
      error: err.toString()
    };
  }
}

function legacyVerificarAccesoUsuario_(emailOpcional) {
  var roleInfo = getCurrentUserRole(emailOpcional);
  if (roleInfo && roleInfo.registered && roleInfo.active && roleInfo.email) {
    return {
      autorizado: true,
      email: roleInfo.email,
      role: roleInfo.role,
      permissions: roleInfo.permissions
    };
  }
  return {
    autorizado: false,
    email: roleInfo ? roleInfo.email : "",
    role: "denied",
    permissions: {
      Modulo_Dashboard: "No",
      Modulo_POS: "No",
      Modulo_MapaLotes: "No"
    },
    error: roleInfo && roleInfo.error ? roleInfo.error : "Acceso No Autorizado: La cuenta no se encuentra registrada o activa en el sistema."
  };
}

/**
 * Obtiene la lista completa de usuarios registrados en la hoja Usuarios_Autorizados.
 */
function legacyObtenerListaUsuarios_() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { success: false, error: "No se pudo obtener la hoja de cálculo activa." };

    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { success: true, data: [] };

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: true, data: [] };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    var result = [];
    for (var r = 0; r < values.length; r++) {
      var row = values[r];
      var userObj = {};
      var hasData = false;

      for (var c = 0; c < headers.length; c++) {
        var h = headers[c].toString().trim();
        var val = row[c] !== undefined && row[c] !== null ? row[c].toString().trim() : "";
        if (val) hasData = true;
        userObj[h] = val;
      }

      if (hasData) {
        var uEmail = (userObj.Correo_Electronico || userObj.Email || userObj.Correo || "").toLowerCase();
        result.push({
          id: userObj.ID || ("USR-" + String(r + 1).padStart(3, '0')),
          email: uEmail,
          nombre: userObj.Nombre || (uEmail ? uEmail.split('@')[0] : ""),
          rol: (userObj.Rol || userObj.Role || "view").toLowerCase() === "admin" ? "admin" : "view",
          estado: (userObj.Estado || userObj.Status || "ACTIVO").toUpperCase(),
          moduloDashboard: userObj.Modulo_Dashboard || userObj.Modulo_Inicio || "Sí",
          moduloPOS: userObj.Modulo_POS || userObj.Modulo_Gastos || "Sí",
          moduloMapaLotes: userObj.Modulo_MapaLotes || userObj.Modulo_Lotes || "Sí"
        });
      }
    }

    return { success: true, data: result };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

/**
 * Agrega o actualiza un usuario en la hoja Usuarios_Autorizados.
 */
function legacyGuardarUsuario_(datos) {
  try {
    var email = datos.email || datos.Email || datos.Correo_Electronico || "";
    var nombre = datos.nombre || datos.Nombre || (email ? email.split('@')[0] : "");
    var rol = (datos.rol || datos.Rol || "view").toLowerCase() === "admin" ? "admin" : "view";
    var estado = (datos.estado || datos.Estado || "ACTIVO").toUpperCase();

    var modDashboard = datos.Modulo_Dashboard || datos.moduloDashboard || "Sí";
    var modPOS = datos.Modulo_POS || datos.moduloPOS || "Sí";
    var modMapaLotes = datos.Modulo_MapaLotes || datos.moduloMapaLotes || "Sí";

    email = email.trim().toLowerCase();
    if (!email) return { success: false, error: "El correo electrónico es obligatorio." };

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { success: false, error: "No se encontró la hoja de cálculo." };

    var sheet = ss.getSheetByName(HOJA_USUARIOS);

    if (!sheet) {
      sheet = ss.insertSheet(HOJA_USUARIOS);
      sheet.appendRow(HEADERS_USUARIOS);
    }
    if (typeof asegurarColumnas === "function") {
      asegurarColumnas(sheet, HEADERS_USUARIOS);
    }

    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    var colMap = {};
    for (var c = 0; c < headers.length; c++) {
      colMap[headers[c].toString().trim()] = c + 1; // 1-based index
    }

    var emailRowIdx = -1;
    if (lastRow >= 2) {
      var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
      var emailColIdx = (colMap["Correo_Electronico"] || colMap["Email"] || colMap["Correo"] || 2) - 1;

      for (var r = 0; r < values.length; r++) {
        var rowEmail = values[r][emailColIdx] ? values[r][emailColIdx].toString().trim().toLowerCase() : "";
        if (rowEmail === email) {
          emailRowIdx = r + 2;
          break;
        }
      }
    }

    var rowValuesMap = {
      "Correo_Electronico": email,
      "Nombre": nombre,
      "Rol": rol,
      "Estado": estado,
      "Modulo_Dashboard": modDashboard,
      "Modulo_POS": modPOS,
      "Modulo_MapaLotes": modMapaLotes
    };

    if (emailRowIdx !== -1) {
      // Actualizar usuario existente columna por columna según encabezados
      for (var key in rowValuesMap) {
        if (colMap[key]) {
          sheet.getRange(emailRowIdx, colMap[key]).setValue(rowValuesMap[key]);
        }
      }
      return { success: true, message: "Usuario " + email + " actualizado correctamente." };
    } else {
      // Insertar nuevo usuario armando fila según HEADERS_USUARIOS
      var nextNum = lastRow < 2 ? 1 : lastRow;
      var idUsr = "USR-" + String(nextNum).padStart(3, '0');
      rowValuesMap["ID"] = idUsr;

      var newRow = [];
      for (var i = 0; i < HEADERS_USUARIOS.length; i++) {
        var head = HEADERS_USUARIOS[i];
        newRow.push(rowValuesMap[head] !== undefined ? rowValuesMap[head] : "Sí");
      }
      sheet.appendRow(newRow);
      return { success: true, message: "Usuario " + email + " registrado con éxito." };
    }
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

/**
 * Elimina un usuario de la hoja Usuarios_Autorizados por correo electrónico.
 */
function legacyEliminarUsuario_(email) {
  try {
    var targetEmail = String(email || "").trim().toLowerCase();
    if (!targetEmail) return { success: false, error: "Especifica un correo válido." };

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { success: false, error: "No existe la hoja de cálculo." };

    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { success: false, error: "La hoja Usuarios no existe." };

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: false, error: "No hay usuarios registrados." };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    var emailColIdx = -1;
    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim().toLowerCase();
      if (h === "correo_electronico" || h === "email" || h === "correo") {
        emailColIdx = c;
        break;
      }
    }
    if (emailColIdx === -1) emailColIdx = 1;

    for (var r = 0; r < values.length; r++) {
      var rowEmail = values[r][emailColIdx] ? values[r][emailColIdx].toString().trim().toLowerCase() : "";
      if (rowEmail === targetEmail) {
        sheet.deleteRow(r + 2);
        return { success: true, message: "Usuario " + targetEmail + " eliminado correctamente." };
      }
    }

    return { success: false, error: "No se encontró el usuario con correo " + targetEmail };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}
